import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.datasets import fetch_openml
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score

USE_LOCAL_CSV = False
LOCAL_CSV_PATH = "fashion_mnist.csv"
TARGET_COLUMN = "label"

if USE_LOCAL_CSV:
    print("Importing local Fashion-MNIST CSV:", LOCAL_CSV_PATH)
    df = pd.read_csv(LOCAL_CSV_PATH)
    y = df[TARGET_COLUMN].to_numpy()
    X = df.drop(columns=[TARGET_COLUMN]).to_numpy(dtype="float32") / 255.0
else:
    print("Loading Fashion-MNIST dataset...")
    data = fetch_openml("Fashion-MNIST", version=1, as_frame=False, parser="auto")
    X, y = data.data.astype("float32") / 255.0, data.target.astype(int)
print("Dataset shape:", X.shape, "Classes:", np.unique(y))
X_train, X_test, y_train, y_test = train_test_split(X, y, train_size=min(12000, int(len(X)*0.8)), test_size=min(3000, int(len(X)*0.2)), random_state=42, stratify=y)
print("Training samples:", len(X_train), "Testing samples:", len(X_test))
scores = []
for k in [1, 3, 5, 7, 9]:
    print(f"Training KNN with K={k}...")
    model = KNeighborsClassifier(n_neighbors=k, n_jobs=-1).fit(X_train, y_train)
    pred = model.predict(X_test)
    score = accuracy_score(y_test, pred)
    scores.append(score)
    print("K:", k, "Accuracy:", score, "Weights: not applicable for KNN.")
plt.plot([1, 3, 5, 7, 9], scores, marker="o")
plt.xlabel("K")
plt.ylabel("Accuracy")
plt.tight_layout()
plt.show()
