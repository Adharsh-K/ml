import numpy as np
import pandas as pd
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score, f1_score, log_loss

USE_LOCAL_CSV = False
LOCAL_CSV_PATH = "breast_cancer.csv"
TARGET_COLUMN = "target"

if USE_LOCAL_CSV:
    print("Importing local CSV:", LOCAL_CSV_PATH)
    df = pd.read_csv(LOCAL_CSV_PATH).dropna()
    if TARGET_COLUMN not in df.columns:
        TARGET_COLUMN = "diagnosis" if "diagnosis" in df.columns else df.columns[-1]
    y = df[TARGET_COLUMN]
    if y.dtype == object:
        y = pd.factorize(y)[0]
    X = df.drop(columns=[TARGET_COLUMN]).select_dtypes(include=np.number)
else:
    print("Loading Breast Cancer Wisconsin dataset...")
    data = load_breast_cancer(as_frame=True)
    X, y = data.data, data.target
print("Dataset shape:", X.shape)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, stratify=y, random_state=42)
print("Training samples:", len(X_train), "Testing samples:", len(X_test))
models = {
    "MLE": make_pipeline(StandardScaler(), LogisticRegression(penalty=None, solver="lbfgs", max_iter=5000)),
    "MAP L2": make_pipeline(StandardScaler(), LogisticRegression(penalty="l2", C=1.0, solver="lbfgs", max_iter=5000)),
    "MAP L1": make_pipeline(StandardScaler(), LogisticRegression(penalty="l1", C=1.0, solver="liblinear", max_iter=5000))
}
for name, model in models.items():
    print("Training", name, "model...")
    model.fit(X_train, y_train)
    pred, prob = model.predict(X_test), model.predict_proba(X_test)
    lr = model.named_steps["logisticregression"]
    print(name, "Accuracy:", accuracy_score(y_test, pred), "F1:", f1_score(y_test, pred), "Log loss:", log_loss(y_test, prob))
    print("Final weights:", lr.coef_)
    print("Final intercept:", lr.intercept_, "Nonzero coefficients:", np.count_nonzero(lr.coef_))
