import numpy as np
import pandas as pd
from sklearn.datasets import fetch_openml
from sklearn.model_selection import train_test_split
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVC, LinearSVC
from sklearn.metrics import accuracy_score, f1_score

USE_LOCAL_CSV = False
LOCAL_CSV_PATH = "fashion_mnist.csv"
TARGET_COLUMN = "label"

if USE_LOCAL_CSV:
    print("Importing local Fashion-MNIST CSV:", LOCAL_CSV_PATH)
    df = pd.read_csv(LOCAL_CSV_PATH)
    y = df[TARGET_COLUMN].to_numpy()
    X = df.drop(columns=[TARGET_COLUMN]).to_numpy(dtype="float32") / 255.0
else:
    print("Loading Fashion-MNIST dataset...")
    data = fetch_openml("Fashion-MNIST", version=1, as_frame=False, parser="auto")
    X, y = data.data.astype("float32") / 255.0, data.target.astype(int)
print("Dataset shape:", X.shape)
X_train, X_test, y_train, y_test = train_test_split(X, y, train_size=min(5000, int(len(X)*0.8)), test_size=min(1500, int(len(X)*0.2)), random_state=42, stratify=y)
print("Training samples:", len(X_train), "Testing samples:", len(X_test))
models = {"Linear": make_pipeline(StandardScaler(), LinearSVC(C=1.0, max_iter=5000, dual="auto")), "Polynomial": make_pipeline(StandardScaler(), SVC(kernel="poly", degree=3, C=1.0)), "RBF": make_pipeline(StandardScaler(), SVC(kernel="rbf", C=1.0, gamma="scale"))}
for name, model in models.items():
    print("Training SVM with", name, "kernel...")
    model.fit(X_train, y_train)
    pred = model.predict(X_test)
    print(name, "Accuracy:", accuracy_score(y_test, pred), "Macro F1:", f1_score(y_test, pred, average="macro"))
    estimator = model.named_steps[list(model.named_steps.keys())[-1]]
    if hasattr(estimator, "coef_"):
        print("Final weights:", estimator.coef_)
    else:
        print("Final weights: not directly available for this kernel; support vectors:", estimator.support_vectors_.shape)
